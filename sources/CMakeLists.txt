cmake_minimum_required(VERSION 3.20)
project(MindMapPortable VERSION 3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(mindmap_inflate STATIC vendor/miniz/miniz_tinfl.c)
set_source_files_properties(vendor/miniz/miniz_tinfl.c PROPERTIES LANGUAGE CXX)
target_compile_definitions(mindmap_inflate PUBLIC MINIZ_NO_ZLIB_APIS MINIZ_NO_ARCHIVE_APIS)
if(WIN32)
  enable_language(RC)
  add_executable(MindMap WIN32 src/app.cpp assets/app.rc)
  target_compile_definitions(MindMap PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(MindMap PRIVATE mindmap_inflate gdiplus comdlg32 comctl32 shell32 user32 gdi32 ole32 cabinet dwmapi)
  if(MINGW)
    target_compile_options(MindMap PRIVATE -Wall -Wextra -O2)
    target_link_options(MindMap PRIVATE -static -static-libgcc -static-libstdc++ -municode -s)
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_compile_definitions(core_tests PRIVATE NOMINMAX)
  if(WIN32)
    target_link_libraries(core_tests PRIVATE cabinet)
  endif()
  if(MINGW)
    target_compile_options(core_tests PRIVATE -O1 -Wall -Wextra)
    target_link_options(core_tests PRIVATE -static)
  endif()
  add_test(NAME core_and_documents COMMAND core_tests)
  add_executable(workspace_tests tests/workspace_tests.cpp)
  target_compile_definitions(workspace_tests PRIVATE NOMINMAX)
  target_link_libraries(workspace_tests PRIVATE mindmap_inflate)
  add_test(NAME workspace_media_imports COMMAND workspace_tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/compressed.xmind")
endif()
